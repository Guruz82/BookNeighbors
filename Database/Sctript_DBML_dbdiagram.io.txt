
Project BookNeighbors {
  database_type: 'PostgreSQL'
}

Table AspNetUsers {
  Id uuid [pk]
  UserName varchar(256)
  NormalizedUserName varchar(256)
  DisplayName varchar(100) [not null]
  Bio text
  PrivacyNoticeAcknowledged boolean [not null]
  PrivacyNoticeAcknowledgedAt timestamptz
  Email varchar(256)
  NormalizedEmail varchar(256)
  EmailConfirmed boolean [not null]
  PasswordHash text
  SecurityStamp text
  ConcurrencyStamp text
  PhoneNumber text
  PhoneNumberConfirmed boolean [not null]
  TwoFactorEnabled boolean [not null]
  LockoutEnd timestamptz
  LockoutEnabled boolean [not null]
  AccessFailedCount int [not null]
  CreatedAt timestamptz [not null]
  UpdatedAt timestamptz
}

Table Libraries {
  Id uuid [pk]
  user_id uuid [not null, unique]
  Name varchar(150) [not null]
  City varchar(100)
  AreaName varchar(150)
  Location geography(Point,4326) [not null]
  LocationConsent boolean [not null, default: false]
  IsPublished boolean [not null, default: false]
  CreatedAt timestamptz [not null]
  UpdatedAt timestamptz

  indexes {
    Location [name: 'IX_Libraries_Location', type: gist]
  }
}

Table Books {
  Id uuid [pk]
  library_id uuid [not null]
  Title varchar(255) [not null]
  Author varchar(200) [not null]
  PublicationYear int
  ISBN varchar(17)
  Description text
  ImageUrl varchar(1024)
  ThumbnailUrl varchar(1024)
  IsPublished boolean [not null, default: false]
  CreatedAt timestamptz [not null]
  UpdatedAt timestamptz
}

Table Categories {
  Id uuid [pk]
  Name varchar(100) [not null, unique]
}

Table BookCategories {
  BookId uuid [not null]
  CategoryId uuid [not null]

  indexes {
    (BookId, CategoryId) [pk]
  }
}

Table LoanRequestStatuses {
  Id uuid [pk]
  Name varchar(50) [not null, unique]
  CreatedAt timestamptz [not null]
  UpdatedAt timestamptz
}

Table LoanRequests {
  Id uuid [pk]
  book_id uuid [not null]
  requester_id uuid [not null]
  status_id uuid [not null]
  DueDate timestamptz
  CreatedAt timestamptz [not null]
  UpdatedAt timestamptz
}

Table LoanMessages {
  Id uuid [pk]
  LoanRequestId uuid [not null]
  SenderId uuid [not null]
  Text varchar(2000) [not null]
  CreatedAt timestamptz [not null]
}

Table BookViews {
  Id uuid [pk]
  book_id uuid [not null]
  ViewedAt timestamptz [not null]
}

Table Notifications {
  Id uuid [pk]
  UserId uuid [not null]
  Title text [not null]
  Message text [not null]
  Url text
  IsRead boolean [not null]
  CreatedAt timestamptz [not null]
  ReadAt timestamptz
}

Ref AspNetUsers_Libraries:
  Libraries.user_id > AspNetUsers.Id

Ref Libraries_Books:
  Books.library_id > Libraries.Id

Ref Books_Categories_Book:
  BookCategories.BookId > Books.Id

Ref Books_Categories_Category:
  BookCategories.CategoryId > Categories.Id

Ref LoanRequests_Books:
  LoanRequests.book_id > Books.Id

Ref LoanRequests_Users:
  LoanRequests.requester_id > AspNetUsers.Id

Ref LoanRequests_Status:
  LoanRequests.status_id > LoanRequestStatuses.Id

Ref BookViews_Books:
  BookViews.book_id > Books.Id

Ref LoanMessages_LoanRequests:
  LoanMessages.LoanRequestId > LoanRequests.Id

Ref LoanMessages_Users:
  LoanMessages.SenderId > AspNetUsers.Id

Ref Notifications_Users:
  Notifications.UserId > AspNetUsers.Id
